StartAndEndDates:
LOAD
	MIN("%KEY_DATE") 		AS StartDate,
    MAX("%KEY_DATE")	 	AS EndDate
RESIDENT F_PRICES;


LET v.FirstDate = Num(Peek('StartDate', 0, 'StartAndEndDates'));
LET v.LastDate = Num(Peek('EndDate', 0, 'StartAndEndDates'));


//DAY granularity
TempCal:
LOAD
	Date($(v.FirstDate) + RowNo() - 1) AS TempDate
AUTOGENERATE
	$(v.LastDate) - $(v.FirstDate) + 1;


MasterCalendar:
LOAD
	TempDate AS %KEY_DATE,
	WEEK(TempDate) AS "Calendar Week",
	YEAR(TempDate) AS "Calendar Year",
	MONTH(TempDate) AS "Calendar Month",
	DAY(TempDate) AS "Calendar Day",
	WEEKDAY(TempDate) AS Weekday,
	'Q' & CEIL(MONTH(TempDate) / 3) AS Quarter,
	DATE(MONTHSTART(TempDate), 'MMM-YYYY') AS MonthYear,
	WEEK(TempDate) & '-' & YEAR(TempDate) AS WeekYear
RESIDENT TempCal;


DROP TABLE StartAndEndDates;
DROP TABLE TempCal;
