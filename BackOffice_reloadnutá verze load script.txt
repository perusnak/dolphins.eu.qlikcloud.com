//Original data is aggregated by months so Master Calendar has to be in month granularity.

StartAndEndDates:
LOAD
	MIN("%KEY_DATE") 		AS FirstOrderDate,
    MAX("%KEY_DATE")	 	AS LastOrderDate
RESIDENT F_DATA;


LET v.FirstDate = Num(Peek('FirstOrderDate', 0, 'StartAndEndDates'));
LET v.LastDate = Num(Peek('LastOrderDate', 0, 'StartAndEndDates'));


//Generates 12 rows for every year: 1st Jan, 1st Feb, 1st Mar etc...  
TempCal:
LOAD
	AddMonths(Date($(v.FirstDate)), RowNo() -1) AS TempDate
AUTOGENERATE
	((Year($(v.LastDate))*12)+month($(v.LastDate))) - ((Year($(v.FirstDate))*12)+month($(v.FirstDate))) + 1;  //Calculate number of months between two dates


//Common Master Calendar in month granularity (Calendar Day is always 1)
MasterCalendar:
LOAD
	TempDate AS %KEY_DATE,
	WEEK(TempDate) AS "Calendar Week",
	YEAR(TempDate) AS "Calendar Year",
	MONTH(TempDate) AS "Calendar Month",
    num(MONTH(TempDate)) AS "Calendar Month Number",
	DAY(TempDate) AS "Calendar Day",
	WEEKDAY (TempDate) AS Weekday,
	'Q' & CEIL(MONTH(TempDate) / 3) AS Quarter,
	DATE(MONTHSTART(TempDate), 'MMM-YYYY') AS MonthYear,
	WEEK(TempDate) & '-' & YEAR(TempDate) AS WeekYear
RESIDENT TempCal;


DROP TABLE StartAndEndDates;
DROP TABLE TempCal;
